	var nm: String = a[2] if a.size() > 2 else "rosso"

	if _G.kira:
		_G.spr = ArtLibrary.battle_sprite("kira")
		_G.por = ArtLibrary.portrait("kira", "neutral")
		_G.cut = ArtLibrary.cutin("kira")

		_G.boss_name = "DISTRIBUTORE AUTOMATICO"
	if a.size() > 4 and a[4] == "boss2":
		_G.boss_name = "DISTRIBUTORE AUTOMATICO NOTTURNO"
	_G.label = "ULTIMATE · TEMPESTA DI LAME VIOLA" if (a.size() > 3 and a[3] == "ultlong") else ""

		nm = "viola"
	_G.turn_txt = "3/20" if selected else "3"
	_G.pick = "ULTIMATE" if (a.size() > 3 and (a[3] == "ult" or a[3] == "ultlong")) else "SKILL"

		if k != nm:
			_G.others.append(_G.PAL[k])
	_run(a[0] if a.size() > 0 else "/tmp")

const FACADE: String = """shader_type spatial;
render_mode unshaded;
uniform vec3 wall = vec3(0.07, 0.09, 0.17);
uniform vec3 fogc = vec3(0.06, 0.13, 0.20);
uniform float seed = 1.0;
varying float d;
varying vec3 wp;
float h(vec2 p) { return fract(sin(dot(p, vec2(12.9898, 78.233)) + seed) * 43758.5453); }
void vertex() { d = -(MODELVIEW_MATRIX * vec4(VERTEX, 1.0)).z; wp = (MODEL_MATRIX * vec4(VERTEX, 1.0)).xyz; }
void fragment() {
	vec2 g = vec2(wp.x + wp.z, wp.y) * vec2(0.9, 0.7);
	vec2 c = floor(g);
	vec2 f = fract(g);
	float win = step(0.18, f.x) * step(f.x, 0.72) * step(0.22, f.y) * step(f.y, 0.78);
	float on = step(0.62, h(c));
	vec3 col = wall * (0.8 + 0.4 * h(c + 3.0));
	col = mix(col, vec3(1.0, 0.78, 0.45) * (0.7 + 0.5 * h(c + 7.0)), win * on);
	col = mix(col, vec3(0.03, 0.05, 0.10), win * (1.0 - on) * 0.7);
	float fog = 1.0 - exp(-d * 0.035);
	ALBEDO = mix(col, fogc, fog);
}"""
const COBBLE: String = """shader_type spatial;
render_mode unshaded;
uniform vec3 fogc = vec3(0.06, 0.13, 0.20);
varying float d;
varying vec3 wp;
float h(vec2 p) { return fract(sin(dot(p, vec2(12.9898, 78.233))) * 43758.5453); }
void vertex() { d = -(MODELVIEW_MATRIX * vec4(VERTEX, 1.0)).z; wp = (MODEL_MATRIX * vec4(VERTEX, 1.0)).xyz; }
void fragment() {
	vec2 g = wp.xz * vec2(1.6, 2.4);
	g.x += floor(g.y) * 0.5;
	vec2 c = floor(g);
	vec2 f = fract(g);
	float gap = step(f.x, 0.07) + step(f.y, 0.09);
	vec3 stone = vec3(0.20, 0.21, 0.27) * (0.65 + 0.7 * h(c));
	// luci calde dei lampioni sul selciato
	float l1 = exp(-length(wp.xz - vec2(-3.4, -4.0)) * 0.26);
	float l2 = exp(-length(wp.xz - vec2(4.6, -9.0)) * 0.22);
	float l3 = exp(-length(wp.xz - vec2(-4.8, -15.0)) * 0.2);
	stone += vec3(1.0, 0.62, 0.3) * (l1 + l2 * 0.8 + l3 * 0.6) * 0.34;
	stone = mix(stone, vec3(0.03, 0.04, 0.07), clamp(gap, 0.0, 1.0) * 0.8);
	float fog = 1.0 - exp(-d * 0.045);
	ALBEDO = mix(stone, fogc, fog);
}"""


func _world() -> Control:
	var holder := Control.new()
	holder.size = Vector2(1280, 720)
	var svc := SubViewportContainer.new()
	svc.stretch = true
	svc.size = Vector2(1280, 720)
	holder.add_child(svc)
	var vp := SubViewport.new()
	vp.size = Vector2i(1280, 720)
	vp.own_world_3d = true
	svc.add_child(vp)
	var env := Environment.new()
	env.background_mode = Environment.BG_COLOR
	env.background_color = Color("#01040a")
	env.glow_enabled = true
	env.glow_intensity = 0.3
	env.glow_bloom = 0.0
	var we := WorldEnvironment.new()
	we.environment = env
	vp.add_child(we)
	var cam := Camera3D.new()
	cam.fov = 46.0
	vp.add_child(cam)
	cam.look_at_from_position(Vector3(-0.6, 1.15, 7.5), Vector3(1.4, 1.55, 0.0))
	var floor_m := MeshInstance3D.new()
	var pm := PlaneMesh.new()
	pm.size = Vector2(60, 90)
	floor_m.mesh = pm
	floor_m.position = Vector3(0, 0, -25)
	var cm := ShaderMaterial.new()
	cm.shader = Shader.new()
	cm.shader.code = COBBLE
	floor_m.material_override = cm
	vp.add_child(floor_m)
	# facciate laterali (case) con finestre accese
	var rng := RandomNumberGenerator.new()
	rng.seed = 5
	for i in 9:
		for side in [-1.0, 1.0]:
			var b := MeshInstance3D.new()
			var bm := BoxMesh.new()
			bm.size = Vector3(rng.randf_range(4.0, 6.0), rng.randf_range(7.0, 14.0), rng.randf_range(5.0, 8.0))
			b.mesh = bm
			b.position = Vector3(side * (9.0 + bm.size.x * 0.5 + rng.randf_range(0.0, 1.5)), bm.size.y * 0.5, 6.0 - float(i) * 6.5)
			var fm := ShaderMaterial.new()
			fm.shader = Shader.new()
			fm.shader.code = FACADE
			fm.set_shader_parameter("seed", float(i) * 3.7 + (1.0 if side > 0.0 else 0.0))
			b.material_override = fm
			vp.add_child(b)
	# fondale: edifici lontani + torre
	for i in 6:
		var b2 := MeshInstance3D.new()
		var bm2 := BoxMesh.new()
		bm2.size = Vector3(rng.randf_range(5.0, 9.0), rng.randf_range(8.0, 22.0), 4.0)
		b2.mesh = bm2
		b2.position = Vector3(-18.0 + float(i) * 7.0, bm2.size.y * 0.5, -48.0)
		var fm2 := ShaderMaterial.new()
		fm2.shader = Shader.new()
		fm2.shader.code = FACADE
		fm2.set_shader_parameter("seed", float(i) * 5.1)
		b2.material_override = fm2
		vp.add_child(b2)
	# lampioni: asta + lampada emissiva (con bagliore dal glow)
	for pos in [Vector3(-3.4, 0, -4.0), Vector3(4.6, 0, -9.0), Vector3(-4.8, 0, -15.0), Vector3(5.2, 0, -21.0)]:
		var pole := MeshInstance3D.new()
		var cyl := CylinderMesh.new()
		cyl.top_radius = 0.07
		cyl.bottom_radius = 0.09
		cyl.height = 4.6
		pole.mesh = cyl
		pole.position = pos + Vector3(0, 2.3, 0)
		var pmat := StandardMaterial3D.new()
		pmat.albedo_color = Color("#3a4468")
		pmat.shading_mode = BaseMaterial3D.SHADING_MODE_UNSHADED
		pole.material_override = pmat
		vp.add_child(pole)
		var lamp := MeshInstance3D.new()
		var sph := SphereMesh.new()
		sph.radius = 0.28
		sph.height = 0.56
		lamp.mesh = sph
		lamp.position = pos + Vector3(0, 4.7, 0)
		var lm := StandardMaterial3D.new()
		lm.shading_mode = BaseMaterial3D.SHADING_MODE_UNSHADED
		lm.albedo_color = Color("#ffd9a0")
		lm.emission_enabled = true
		lm.emission = Color("#ffb060")
		lm.emission_energy_multiplier = 3.0
		lamp.material_override = lm
		vp.add_child(lamp)
	# luna
	var moon := MeshInstance3D.new()
	var ms := SphereMesh.new()
	ms.radius = 2.2
	ms.height = 4.4
	moon.mesh = ms
	moon.position = Vector3(8.0, 20.0, -60.0)
	var mm := StandardMaterial3D.new()
	mm.shading_mode = BaseMaterial3D.SHADING_MODE_UNSHADED
	mm.albedo_color = Color("#dfe9ff")
	mm.emission_enabled = true
	mm.emission = Color("#9fb8ff")
	mm.emission_energy_multiplier = 0.8
	moon.material_override = mm
	vp.add_child(moon)
	# sprite 2D
	if selected:
		var dim0 := ColorRect.new()
		dim0.color = Color(0.01, 0.0, 0.05, 0.6)
		dim0.size = Vector2(1280, 720)
		holder.add_child(dim0)
	var sprites := Control.new()
	sprites.size = Vector2(1280, 720)
	holder.add_child(sprites)
	var sh := _Shadow.new()
	sh.size = Vector2(1280, 720)
	sprites.add_child(sh)
	if true:
		var back := _Wheel.new()
		back.size = Vector2(1280, 720)
		back.part = "back"
		back.glow = selected
		back.picked = _G.pick if selected else ""
		sprites.add_child(back)
	var boss := _Boss.new()
	boss.position = Vector2(690, 100)
	boss.size = Vector2(560, 500)
	if selected:
		boss.modulate = Color(0.42, 0.42, 0.55)
	sprites.add_child(boss)
	var hero := _Hero.new()
	hero.position = Vector2(176, 190)
	hero.size = Vector2(420, 450)
	if selected:
		hero.modulate = Color(0.5, 0.5, 0.62)
	sprites.add_child(hero)
	var vig := _Vignette.new()
	vig.size = Vector2(1280, 720)
	holder.add_child(vig)
	return holder

		draw_colored_polygon(PackedVector2Array([a + n * w0 * 0.5, b + n * w1 * 0.5, b - n * w1 * 0.5, a - n * w0 * 0.5]), col)

# ---------- Utilita' di disegno ----------
class _U:
	# Scala unica (px alla risoluzione di progetto 1152x648; la UI scala in proporzione):
	const T_TITLE: int = 28      # titoli grandi (banner, boss)
	const T_CMD: int = 18        # comandi e nomi
	const T_QUICK: int = 15      # comandi rapidi (compatti)
	const T_SUB: int = 14        # sottotitoli / etichette piccole (minimo assoluto)
	const ICON_K: float = 0.6    # icona = 60% del diametro del suo cerchio, ovunque

	static func skew_text(c: CanvasItem, p: Vector2, t: String, sz: int, col: Color, outline: Color = Color(0, 0, 0, 0.9), f: Font = null, ow: int = 8, sk: float = -0.22) -> void:
		var ff: Font = f if f != null else BattleTheme.display()
		c.draw_set_transform_matrix(Transform2D(Vector2(1, 0), Vector2(sk, 1), p))
		if ow > 0:
			c.draw_string_outline(ff, Vector2.ZERO, t, HORIZONTAL_ALIGNMENT_LEFT, -1, sz, ow, outline)
		c.draw_string(ff, Vector2.ZERO, t, HORIZONTAL_ALIGNMENT_LEFT, -1, sz, col)
		c.draw_set_transform_matrix(Transform2D.IDENTITY)

	static func text_w(t: String, sz: int, f: Font = null) -> float:
		var ff: Font = f if f != null else BattleTheme.display()
		return ff.get_string_size(t, HORIZONTAL_ALIGNMENT_LEFT, -1, sz).x

	static func icon(c: CanvasItem, key: String, center: Vector2, px: float, col: Color) -> void:
		var tex: Texture2D = GameIcon.texture_for(GameIcon.resolve(key), "", px)
		if tex != null:
			c.draw_texture_rect(tex, Rect2(center - Vector2(px, px) * 0.5, Vector2(px, px)), false, col)

	static func slant(r: Rect2, sl: float) -> PackedVector2Array:
		return PackedVector2Array([Vector2(r.position.x + sl, r.position.y), Vector2(r.end.x, r.position.y), Vector2(r.end.x - sl, r.end.y), Vector2(r.position.x, r.end.y)])

	# Pennellata: striscia a larghezza variabile con bordi irregolari
	static func brush(c: CanvasItem, a: Vector2, b: Vector2, w0: float, w1: float, col: Color, sd: int, bend: float = 0.0, jag: float = 0.22, n: int = 16) -> void:
		var rng := RandomNumberGenerator.new()
		rng.seed = sd
		var d: Vector2 = b - a
		var nrm: Vector2 = Vector2(-d.y, d.x).normalized()
		var L := PackedVector2Array()
		var Rr := PackedVector2Array()
		for i in n + 1:
			var t: float = float(i) / float(n)
			var p: Vector2 = a + d * t + nrm * sin(t * PI) * bend
			var w: float = maxf(1.5, lerpf(w0, w1, t) * 0.5)
			L.append(p + nrm * w * (1.0 + rng.randf_range(-jag, jag)))
			Rr.append(p - nrm * w * (1.0 + rng.randf_range(-jag, jag)))
		for i in n:
			c.draw_colored_polygon(PackedVector2Array([L[i], L[i + 1], Rr[i + 1], Rr[i]]), col)
		# setole a secco oltre la fine
		for k in 4:
			var o: Vector2 = nrm * (float(k) - 1.5) * w1 * 0.3
			c.draw_line(b + o, b + o + d.normalized() * rng.randf_range(14.0, 44.0), col, 2.0)

	static func star(c: CanvasItem, p: Vector2, r: float, col: Color) -> void:
		var pts := PackedVector2Array()
		for k in 16:
			var a: float = -PI / 2.0 + TAU * float(k) / 16.0
			pts.append(p + Vector2(cos(a), sin(a)) * (r if k % 2 == 0 else r * 0.58))
		c.draw_colored_polygon(pts, col)
		pts.append(pts[0])
		c.draw_polyline(pts, Color(0.02, 0.03, 0.06, 0.9), 2.0)

	static func head(c: CanvasItem, p: Vector2, rad: float, hair: Color) -> void:
		c.draw_circle(p, rad, Color("#0a0b14"))
		c.draw_circle(p + Vector2(0, rad * 0.12), rad * 0.5, Color("#f4cdb8"))
		c.draw_colored_polygon(PackedVector2Array([p + Vector2(-rad * 0.62, rad * 0.05), p + Vector2(-rad * 0.3, -rad * 0.66), p + Vector2(rad * 0.34, -rad * 0.62), p + Vector2(rad * 0.64, rad * 0.02), p + Vector2(rad * 0.3, -rad * 0.18), p + Vector2(-rad * 0.3, -rad * 0.16)]), hair)

class _Hud extends Control:

	func _poly_line(pts: PackedVector2Array, col: Color, w: float) -> void:
		var p2 := PackedVector2Array(pts)
		p2.append(pts[0])
		draw_polyline(p2, col, w)

	# Barra sottile: binario scuro con filo chiaro, riempimento con luce in alto, punta luminosa e segmenti a taglio fine.
	func _thin(p: Vector2, w: float, frac: float, col: Color, h: float, segs: int = 0, tip: bool = true) -> void:
		draw_rect(Rect2(p + Vector2(-1, -1), Vector2(w + 2, h + 2)), Color(0, 0, 0, 0.62))
		draw_rect(Rect2(p, Vector2(w, h)), Color(col, 0.16))
		var fw: float = w * clampf(frac, 0.0, 1.0)
		draw_rect(Rect2(p, Vector2(fw, h)), col)
		draw_rect(Rect2(p, Vector2(fw, maxf(1.0, h * 0.4))), Color(1, 1, 1, 0.35))
		for t in segs:
			var x: float = p.x + w * float(t + 1) / float(segs + 1)
			draw_line(Vector2(x, p.y - 1), Vector2(x, p.y + h + 1), Color(0.02, 0.03, 0.07, 0.9), 1.5)
		if tip and fw > 2.0:
			draw_rect(Rect2(Vector2(p.x + fw - 1.5, p.y - 2.0), Vector2(2.0, h + 4.0)), Color(1, 1, 1, 0.95))
		draw_line(p + Vector2(0, h + 1.5), p + Vector2(w, h + 1.5), Color(1, 1, 1, 0.22), 1.0)

	func _bar(r: Rect2, frac: float, col: Color, ticks: int = 0) -> void:
		draw_colored_polygon(_U.slant(r, 4.0), Color(0, 0, 0, 0.78))
		var fr := Rect2(r.position, Vector2(r.size.x * frac, r.size.y))
		draw_colored_polygon(_U.slant(fr, 4.0), col)
		draw_rect(Rect2(r.position + Vector2(2, 1), Vector2(maxf(0.0, r.size.x * frac - 4.0), r.size.y * 0.32)), Color(1, 1, 1, 0.3))
		for t in ticks:
			var x: float = r.position.x + r.size.x * float(t + 1) / float(ticks + 1)
			draw_line(Vector2(x, r.position.y), Vector2(x, r.end.y), Color(0, 0, 0, 0.75), 2.0)

	# Curva di opacita': 100% all'inizio, 50% a x1, poi discesa dolce (ease-out) fino a 0% in x2.
	func _fade_a(x: float, x0: float, x1: float, x2: float) -> float:
		if x <= x0:
			return 1.0
		if x <= x1:
			return lerpf(1.0, 0.5, (x - x0) / maxf(1.0, x1 - x0))
		if x >= x2:
			return 0.0
		return 0.5 * pow(1.0 - (x - x1) / maxf(1.0, x2 - x1), 1.7)

	# Banner a bande continue con colore per vertice (nessuna cucitura): bordi esterni sfumati sopra, sotto e a sinistra,
	# alone e filetti colorati che sfumano lungo la lunghezza con la stessa curva di opacita'.
	func _band(xs: PackedFloat32Array, y0: float, y1: float, sl: float, col: Color, base_a: float) -> void:
		var pts := PackedVector2Array()
		var cols := PackedColorArray()
		for i in xs.size():
			pts.append(Vector2(xs[i] + sl, y0))
			cols.append(Color(col, base_a * _bw[i]))
		for i in range(xs.size() - 1, -1, -1):
			pts.append(Vector2(xs[i], y1))
			cols.append(Color(col, base_a * _bw[i]))
		draw_polygon(pts, cols)

	var _bw := PackedFloat32Array()

	func _fade_banner(x0: float, x1: float, x2: float, y: float, h: float, sl: float) -> void:
		var xs := PackedFloat32Array()
		_bw = PackedFloat32Array()
		var xl: float = x0 - 12.0
		var x: float = xl
		while x < x2 + 8.0:
			xs.append(x)
			_bw.append(_fade_a(maxf(x, x0), x0, x1, x2) * clampf((x - xl) / 12.0, 0.0, 1.0))
			x += 8.0
		var dark := Color(0.04, 0.05, 0.1)
		_band(xs, y - 12.0, y + h + 12.0, sl, _G.col, 0.14)
		for k in range(6, -1, -1):
			var e: float = float(k) * 1.8
			_band(xs, y - e, y + h + e, sl, dark, 0.20 if k > 0 else 0.55)
		var top_p := PackedVector2Array()
		var bot_p := PackedVector2Array()
		var tc := PackedColorArray()
		var tg := PackedColorArray()
		for i in xs.size():
			top_p.append(Vector2(xs[i] + sl, y))
			bot_p.append(Vector2(xs[i], y + h))
			tc.append(Color(1, 1, 1, _bw[i]))
			tg.append(Color(1, 1, 1, 0.22 * _bw[i]))
		draw_polyline_colors(top_p, tg, 6.0)

func _run(dir: String) -> void:
	var rootc := Control.new()
	rootc.size = Vector2(1280, 720)
	rootc.scale = Vector2(0.9, 0.9)
	root.add_child(rootc)
	rootc.add_child(_world())
	var wheel := _Wheel.new()
	wheel.size = Vector2(1280, 720)
	wheel.light_mode = selected
	wheel.part = "front"
	wheel.picked = _G.pick if selected else ""
	rootc.add_child(wheel)
	var hud := _Hud.new()
	hud.size = Vector2(1280, 720)
	hud.selected = selected
	rootc.add_child(hud)
	await create_timer(1.3).timeout
	root.get_viewport().get_texture().get_image().save_png(dir.path_join("hud26_%s_%s.png" % [_G.name_, ("scelta_" + _G.pick.to_lower()) if selected else "libera"]))
	quit()
